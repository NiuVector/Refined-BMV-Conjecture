\begin{abstract}
Let $\mathcal{A}_{n,m}(A,B) := \binom{n+m}{m}^{-1}\Tr S_{n+m,m}(A,B)$ denote the normalized average trace over all $\binom{n+m}{m}$ matrix words containing $n$ copies of $A$ and $m$ copies of $B$. In 2017, D.~H\"agele proposed the refined Bessis--Moussa--Villani (BMV) conjecture (\href{https://oqp.iqoqi.oeaw.ac.at/open-quantum-problems}{IQOQI Open Quantum Problems \#40} / \href{https://openquantumproblems.com/p/M14}{Open Quantum Problems M14}). While its upper bound was disproved in early 2026, the lower bound $\mathcal{A}_{n,m}(e^\alpha,e^\beta)\ge\Tr(e^{n\alpha+m\beta})$ and its stronger evenly-spread word form $\mathcal{A}_{n,m}(A,B)\ge\Tr((A^{n/m}B)^m)$ (denoted $(E_{n,m})$) remained open for all dimensions $d\ge 4$ whenever $\min(n,m)\ge 3$, and even for $d=3$ whenever $n+m\ge 8$.

In this paper, we prove $(E_{n,m})$ and $(L_{n,m})$ in \textbf{every matrix dimension $d\ge 1$} for:
(1) \textbf{the entire infinite region $\min(n,m)\le 3$} (all integers $n,m\ge 1$ with $\min(n,m)\le 3$), together with the $n\to\infty$ third-order Duhamel trace inequality $\left.\frac{\mathrm{d}^3}{\mathrm{d}t^3}\right|_{t=0}\Tr(e^{H+tB})\ge\Tr((e^{H/3}B)^3)$ and the $n\to 0^+$ logarithmic limit $\Phi_0$; and
(2) \textbf{the first open $m=4$ case $(n,m)=(4,4)$} (as well as $(1,4),(2,4),(3,4)$), via a completely elementary, computer-free $3\times 3 \oplus 2\times 2$ pure integer-entry Gram matrix identity whose Sylvester minors have strictly positive integer coefficients---holding even for \textbf{arbitrary indefinite complex Hermitian $X$}.
The smallest remaining open cases of \href{https://openquantumproblems.com/p/M14}{OQP M14} are thus pushed to $(5,4)/(4,5)$ and $(5,5)$.
\end{abstract}

\takeaway{Main Results (All Dimensions $d \ge 1$)}{
For every dimension $d\ge 1$, all $d\times d$ complex Hermitian $A,B\ge 0$, and all Hermitian $\alpha,\beta,H$:
(i) \textbf{All $\min(n,m)\le 3$}: $\mathcal{A}_{n,m}(A,B)\ge\Tr((A^{n/m}B)^m)$ and $\mathcal{A}_{n,m}(e^\alpha,e^\beta)\ge\Tr(e^{n\alpha+m\beta})$ hold for all $n,m\ge 1$ with $\min(n,m)\le 3$;
(ii) \textbf{Third-order Duhamel ($n\to\infty$)}: $\left.\frac{\mathrm{d}^3}{\mathrm{d}t^3}\right|_{t=0}\Tr(e^{H+tB})\ge\Tr((e^{H/3}B)^3)$;
(iii) \textbf{Case $(4,4)$ (elementary, indefinite $X$)}: $\mathcal{A}_{4,4}(A,B)\ge\Tr((AB)^4)$ holds for all $A>0$ and all $B = A^{-1/2} X A^{-1/2}$ with arbitrary (even indefinite) complex Hermitian $X$, yielding $\mathcal{A}_{4,4}(e^\alpha,e^\beta)\ge\Tr(e^{4\alpha+4\beta})$.}

\section{Introduction and Main Results}

The classical Bessis--Moussa--Villani (BMV) conjecture (1975), resolved by Stahl in 2013, states that $\Tr S_{n+m,m}(A,B)\ge 0$ for all positive semidefinite $A,B\ge 0$ and integers $n,m\ge 0$. In 2017, D.~H\"agele proposed a quantitative refinement for $\mathcal{A}_{n,m}(A,B):=\binom{n+m}{m}^{-1}\Tr S_{n+m,m}(A,B)$ (\href{https://oqp.iqoqi.oeaw.ac.at/open-quantum-problems}{IQOQI Problem \#40}, \href{https://openquantumproblems.com/p/M14}{OQP M14}):
\[
\Tr\!\big(e^{n\alpha+m\beta}\big)\;\le\;\mathcal{A}_{n,m}(e^\alpha,e^\beta)\;\le\;\Tr\!\big(e^{n\alpha}e^{m\beta}\big).
\]
In 2026, Cha--Lee (\href{https://arxiv.org/abs/2603.19927}{arXiv:2603.19927}) and Dinh (\href{https://arxiv.org/html/2605.17782v1}{arXiv:2605.17782}) disproved the upper bound at $d=3,\,(n,m)=(5,5)$ (see also Dinh--Le--Vuong, \href{https://arxiv.org/html/2610.31851v1}{arXiv:2610.31851}). In October 2026, H\"agele (\href{https://arxiv.org/html/2611.00129v1}{arXiv:2611.00129}) established the stronger ``evenly-spread word'' inequality
\begin{equation}\label{eq:Enm}
  \mathcal{A}_{n,m}(A,B) \ge \Tr\!\big((A^{n/m}B)^m\big) \qquad (E_{n,m}),
\end{equation}
which implies $(L_{n,m})$ via Araki--Lieb--Thirring and Lie--Trotter, for $\min(n,m)\le 2$ (all $d$), $d\le 2$ (all $n,m$), and $(n,m)\in\{(3,3),(4,3)\}$ at $d=3$ using computer-assisted bivariate Bernstein certificates (116 and 2,283 terms). All $d\ge 4$ cases with $\min(n,m)\ge 3$ and all $n+m\ge 8$ cases even at $d=3$ remained open.

\begin{theorem}[Complete Resolution of $\min(n,m)\le 3$ in All Dimensions]\label{thm:min3}
For every dimension $d\ge 1$ and all integers $n,m\ge 1$ with $\min(n,m)\le 3$:
\begin{enumerate}
  \item For all $d\times d$ complex Hermitian $A,B\ge 0$, $\mathcal{A}_{n,m}(A,B) \ge \Tr\!\big((A^{n/m}B)^m\big)$.
  \item For all $d\times d$ complex Hermitian $\alpha,\beta$, $\mathcal{A}_{n,m}(e^\alpha,e^\beta)\ge\Tr(e^{n\alpha+m\beta})$, with equality if and only if $[\alpha,\beta]=0$.
\end{enumerate}
\end{theorem}

\begin{theorem}[Third-Order Duhamel Trace Inequality, $n\to\infty$]\label{thm:duhamel}
For every dimension $d\ge 1$, every complex Hermitian $H$, and every complex Hermitian $B\ge 0$,
\begin{equation}\label{eq:duhamel}
  \left.\frac{\mathrm{d}^3}{\mathrm{d}t^3}\right|_{t=0}\Tr\!\big(e^{H+tB}\big)
  = \mathbb{E}_{(U,V,W)\sim\operatorname{Unif}(\Delta_2)}\!\left[\Tr\!\big(e^{UH}Be^{VH}Be^{WH}B\big)\right]
  \ge \Tr\!\big((e^{H/3}B)^3\big),
\end{equation}
with equality for $B>0$ if and only if $[H,B]=0$.
\end{theorem}

\begin{theorem}[Elementary All-Dimension Resolution of $(4,4)$ for Indefinite Hermitian $X$]\label{thm:44}
For every dimension $d\ge 1$, every $d\times d$ complex Hermitian positive definite matrix $A>0$, and every (possibly indefinite) complex Hermitian matrix $X$, setting $B := A^{-1/2} X A^{-1/2}$,
\begin{equation}\label{eq:E44}
  \mathcal{A}_{4,4}(A,B) \ge \Tr\!\big((AB)^4\big) = \Tr(X^4).
\end{equation}
In particular, $(E_{4,4})$ holds for all $A,B\ge 0$ (by continuity) and $\mathcal{A}_{4,4}(e^\alpha,e^\beta)\ge\Tr(e^{4\alpha+4\beta})$ holds for all complex Hermitian $\alpha,\beta$ in every dimension $d\ge 1$.
\end{theorem}

\section{Duality Transfer and the Dimension-Independent Kernel Framework}

\begin{proposition}[Duality + Araki--Lieb--Thirring Transfer]\label{prop:transfer}
For all integers $1\le n\le m$ and every dimension $d\ge 1$, $(E_{m,n})$ implies $(E_{n,m})$ for all $A,B\ge 0$.
\end{proposition}

\begin{proof}
By binomial symmetry $\mathcal{A}_{n,m}(A,B)=\mathcal{A}_{m,n}(B,A)$ and $(E_{m,n})$, we have $\mathcal{A}_{n,m}(A,B) \ge \Tr((B^{m/n}A)^n) = \Tr((A^{1/2}B^{m/n}A^{1/2})^n)$. Applying Araki--Lieb--Thirring with $P=A^{n/m}\ge 0$, $Q=B\ge 0$, $p=n$, and $r=m/n\ge 1$ yields
\[
\Tr\!\big((A^{n/m}B)^m\big) = \Tr\!\big((P^{1/2}QP^{1/2})^{r\cdot n}\big) \le \Tr\!\big((P^{r/2}Q^r P^{r/2})^n\big) = \Tr\!\big((A^{1/2}B^{m/n}A^{1/2})^n\big) \le \mathcal{A}_{n,m}(A,B). \qedhere
\]
\end{proof}

Proposition~\ref{prop:transfer} immediately proves $(E_{2,m})$ for all $m\ge 2$ (from elementary $(E_{m,2})$), and reduces Theorem~\ref{thm:min3} to proving $(E_{n,3})$ for all $n\ge 1$ and all $d\ge 1$.

For general $m$, diagonalizing $A=\operatorname{diag}(a_1,\dots,a_d)>0$ and setting $X:=A^{n/(2m)}BA^{n/(2m)}$ yields
\begin{equation}\label{eq:general_m_sum}
  \mathcal{A}_{n,m}(A,B) - \Tr(X^m) = \sum_{i_1,\dots,i_m=1}^d \Phi_{n,m}(a_{i_1},\dots,a_{i_m})\, X_{i_1 i_2}\cdots X_{i_m i_1},
\end{equation}
where $\Phi_{n,m}(x_1,\dots,x_m) := \binom{n+m-1}{m-1}^{-1}\frac{h_n(x_1,\dots,x_m)}{(x_1\cdots x_m)^{n/m}} - 1$.
\begin{itemize}
  \item \textbf{Odd $m=3$ (Schur product with $X\ge 0$)}: Any symmetric PSD kernel $K_n(u,v)\succeq 0$ with $\sum_{\mathrm{cyc}} K_n(y/x,z/x) = 3\Phi_{n,3}(x,y,z)$ yields $\sum_{i=1}^d (v^{(i)})^*(M^{(i)}\circ X)v^{(i)}\ge 0$ for all $d\ge 1$ by the Schur product theorem ($M^{(i)}_{jk}=K_n(a_j/a_i,a_k/a_i)\succeq 0$, $v^{(i)}_j = X_{ji}$).
  \item \textbf{Even $m=2r$ (antipodal half-path quadratic form, arbitrary Hermitian $X$)}: Fixing $(i_1,i_{r+1})$ pairs forward and backward length-$r$ half-paths $w_p = X_{i_1 i_2}\cdots X_{i_r i_{r+1}}$ and $\overline{w_q} = X_{i_{r+1}i_{r+2}}\cdots X_{i_{2r}i_1}$. Any kernel $G(a_{i_1},a_{i_{r+1}}; a_p; a_q)\succeq 0$ in $(p,q)$ whose $r$-fold antipodal cyclic average equals $\Phi_{n,2r}$ immediately proves $\mathcal{A}_{n,2r}(A,B)\ge\Tr(X^{2r})$ for all $d\ge 1$ and \emph{arbitrary indefinite complex Hermitian $X$}.
\end{itemize}

\section{Elementary All-Dimension Proof of $(4,4)$ and $(1,4),(2,4),(3,4)$}

\begin{proof}[Proof of Theorem~\ref{thm:44}]
Set $x_1=a^2, x_2=b^2, x_3=c^2, x_4=d^2$ ($a,b,c,d>0$). Define the pure integer-entry symmetric block matrices
\[
G_{4,\mathrm{even}}(a,c) := \begin{pmatrix}
2a^8 + 2a^6 c^2 - 3a^4 c^4 + 2a^2 c^6 + 2c^8 & 2(a^3 - c^3)^2 & a^4 - 7a^2 c^2 + c^4 \\
2(a^3 - c^3)^2 & (a-c)^2(9a^2 + 26ac + 9c^2) & 2(a-c)^2 \\
a^4 - 7a^2 c^2 + c^4 & 2(a-c)^2 & 5
\end{pmatrix},
\]
\[
G_{4,\mathrm{odd}}(a,c) := 4\begin{pmatrix}
a^2 c^2 (a^2 + c^2) & -2 a^2 c^2 \\
-2 a^2 c^2 & a^2 + c^2
\end{pmatrix},
\]
and set $\widetilde{K}_4(b,d;a,c) := (1, b^2, b^4) G_{4,\mathrm{even}}(a,c) (1, d^2, d^4)^T + (b, b^3) G_{4,\mathrm{odd}}(a,c) (d, d^3)^T$.
Direct polynomial expansion (verified symbolically to $0$ in SymPy) gives the exact 2-term cyclic identity
\[
\widetilde{K}_4(b,d;a,c) + \widetilde{K}_4(a,c;b,d) \equiv 2\Big(h_4(a^2,b^2,c^2,d^2) - 35 a^2 b^2 c^2 d^2\Big).
\]
Both Gram blocks are positive semidefinite for all $a,c>0$ by Sylvester's criterion on their trailing principal minors:
\begin{itemize}
  \item \textbf{Odd $2\times 2$ block}: $M_1^{\mathrm{odd}} = 4(a^2+c^2) > 0$, $M_2^{\mathrm{odd}} = 16 a^2 c^2 (a^2-c^2)^2 \ge 0$.
  \item \textbf{Even $3\times 3$ block}: $M_1^{\mathrm{even}} = 5 > 0$, $M_2^{\mathrm{even}} = (a-c)^2(41a^2 + 138ac + 41c^2) \ge 0$, and
  \[
  M_3^{\mathrm{even}} = \det G_{4,\mathrm{even}}(a,c) = (a-c)^4\big(61a^8 + 332a^7c + 768a^6c^2 + 1828a^5c^3 + 2582a^4c^4 + 1828a^3c^5 + 768a^2c^6 + 332ac^7 + 61c^8\big) \ge 0,
  \]
  whose coefficients are all strictly positive integers! (At $c=a$, $G_{4,\mathrm{even}}(a,a) = 5(a^4,0,-1)^T(a^4,0,-1)\succeq 0$ and $G_{4,\mathrm{odd}}(a,a) = 8a^2(a^2,-1)^T(a^2,-1)\succeq 0$.)
\end{itemize}
Writing $A = \operatorname{diag}(\alpha_1^2,\dots,\alpha_d^2)$ and $w^{(i_1,i_3)}_j := X_{i_1 j} X_{j i_3}$, cyclic invariance of $X_{i_1 i_2}X_{i_2 i_3}X_{i_3 i_4}X_{i_4 i_1}$ gives the exact manifest sum-of-squares identity for every complex Hermitian $X\in\mathbb{C}^{d\times d}$:
\[
\mathcal{A}_{4,4}(A,B) - \Tr(X^4) = \frac{1}{35}\sum_{i_1,i_3=1}^d \frac{1}{\alpha_{i_1}^2\alpha_{i_3}^2}\Big[\mathbf{u}^{(i_1,i_3)*} G_{4,\mathrm{even}}(\alpha_{i_1},\alpha_{i_3})\mathbf{u}^{(i_1,i_3)} + \mathbf{v}^{(i_1,i_3)*} G_{4,\mathrm{odd}}(\alpha_{i_1},\alpha_{i_3})\mathbf{v}^{(i_1,i_3)}\Big] \ge 0,
\]
where $u^{(i_1,i_3)}_p = \sum_{j=1}^d \alpha_j^{p-2} w^{(i_1,i_3)}_j$ ($p\in\{0,2,4\}$) and $v^{(i_1,i_3)}_q = \sum_{j=1}^d \alpha_j^{q-2} w^{(i_1,i_3)}_j$ ($q\in\{1,3\}$).
\end{proof}

\begin{remark}[Elementary Integer Gram Proofs for $(1,4),(2,4),(3,4)$]
The exact same construction yields computer-free integer Gram proofs for arbitrary indefinite Hermitian $X$ in $(1,4)$, $(2,4)$, and $(3,4)$ (all verified symbolically in \texttt{verify\_m4\_integer\_gram.py} and numerically across complex indefinite Hermitian matrices to relative error $\le 7.6\times 10^{-15}$):
\begin{itemize}
  \item $(1,4)$ ($x_i=\alpha_i^4, \lambda_1=1$): $G_{1,\mathrm{even}}=\bigl(\begin{smallmatrix}a^4-a^2c^2+c^4 & -ac\\ -ac & 1\end{smallmatrix}\bigr)$, $G_{1,\mathrm{odd}}=[(a-c)^2]$.
  \item $(2,4)$ ($x_i=\alpha_i^2, \lambda_2=2$): $G_{2,\mathrm{even}}=\bigl(\begin{smallmatrix}2a^4-a^2c^2+2c^4 & a^2-5ac+c^2\\ a^2-5ac+c^2 & 3\end{smallmatrix}\bigr)$, $G_{2,\mathrm{odd}}=[5(a-c)^2]$.
  \item $(3,4)$ ($x_i=\alpha_i^4, \lambda_3=2$): $4\times 4\oplus 1\times 1$ integer blocks whose trailing minors in $r=(a^2-c^2)^2/(a^2c^2)\ge 0$ are $7$, $a^2c^2(49+35r)$, $a^6c^6(14+575r+175r^2)$, and $a^{10}c^{10}(a^2-c^2)^2(644+7015r+5936r^2+1866r^3+210r^4)>0$.
\end{itemize}
\end{remark}

\section{Complete Resolution of $m=3$ for All $n\ge 1$, $n\to 0^+$, and $n\to\infty$}

\subsection{Explicit All-Dimension Proofs for $n\to 0^+$ and $1\le n\le 6$}
\begin{itemize}
  \item \textbf{Logarithmic limit $n\to 0^+$}: The Loewner integral kernel $K_0(u,v) = \int_0^1 \frac{(u-1)(v-1)}{(tu+1-t)(tv+1-t)}\,t(1-t^2)\,\mathrm{d}t\succeq 0$ satisfies $\sum_{\mathrm{cyc}} K_0(y/x,z/x) = 3\Phi_0(x,y,z)$ identically, proving $\sum_{i,j,k}\Phi_0(a_i,a_j,a_k)X_{ij}X_{jk}X_{ki}\ge 0$ for all $d\ge 1$.
  \item \textbf{Cases $n=1,2$}: $K_1(u,v) = \frac{(u^{1/3}-1)(2u^{1/3}+1)(v^{1/3}-1)(2v^{1/3}+1)}{u^{1/3}v^{1/3}(u^{1/3}+v^{1/3}+1)}\succeq 0$ and pullback $K_2(u,v)=\frac{1}{2}K_1(u^2,v^2)+\frac{1}{2}K_1(u^{-1},v^{-1})\succeq 0$.
  \item \textbf{Cases $n=3,4,6$ (computer-free Frobenius SOS identities)}: Letting $TY+YT=TX-XT$ and $T^{-1}Z+ZT^{-1}=X$ ($Z=\int_0^\infty e^{-sT^{-1}}Xe^{-sT^{-1}}\,\mathrm{d}s\ge 0$), we obtain exact basis-free identities for all $d\ge 1$ and $A>0, B\ge 0$:
  \begin{align*}
    \mathcal{A}_{3,3}(A,B)-\Tr((AB)^3) &= \tfrac{3}{10}\big(\|SX^{1/2}\|_F^2 + 8\|A^{-1/2}RZ^{1/2}\|_F^2\big) \quad (T=A, X=A^{1/2}BA^{1/2}),\\
    \mathcal{A}_{4,3}(A,B)-\Tr((A^{4/3}B)^3) &= \tfrac{1}{5}\|M_1X^{1/2}\|_F^2 + \tfrac{1}{5}\|M_2X^{1/2}\|_F^2 + \|M_3X^{1/2}\|_F^2 + \|YX^{1/2}\|_F^2 + 4\|T^{-1/2}YZ^{1/2}\|_F^2,\\
    \mathcal{A}_{6,3}(A,B)-\Tr((A^2B)^3) &= \tfrac{3}{28}\big(\|H_1X^{1/2}\|_F^2 + 7\|H_2X^{1/2}\|_F^2 + 4\|RX^{1/2}\|_F^2 + 28\|A^{-1/2}G_1Z^{1/2}\|_F^2 + 12\|A^{-1/2}RZ^{1/2}\|_F^2\big),
  \end{align*}
  where $S=AXA^{-1}-X+2R$ ($n=3$), $T=A^{1/3}, X=T^2BT^2, M_1=T^4XT^{-4}+TXT^{-1}-2T^{-2}XT^2, M_2=TXT^{-1}-T^{-2}XT^2, M_3=TXT^{-1}-X+Y$ ($n=4$), and $X=ABA, H_1=A^2XA^{-2}+AXA^{-1}-2X+3R, H_2=X-A^{-1}XA+R, G_1=AXA^{-1}-X-R$ ($n=6$). All three have exact symbolic residual $0$ and equality iff $AB=BA$ via spectral projections on $Z=\int_0^\infty e^{-sT^{-1}}Xe^{-sT^{-1}}\,\mathrm{d}s$.
  \item \textbf{Case $n=5$}: Exact rational two-Cauchy PSD kernel ($\frac{ts}{t+s}$ and $\frac{t^3s^3}{t^3+s^3}$, common denominator $1680$, 19 strictly positive exact rational Sylvester minors, symbolic residual $0$).
\end{itemize}

\subsection{Closure of All $n\ge 7$ and $n=\infty$ via 3D Rigorous Certification}
Setting $\varepsilon := 1/n \in [0,1/7]$ and scaled log-coordinates $s,t>0$ ($v=t-s>0$, $u=e^{\varepsilon s}, v_u=e^{\varepsilon t}$) with $\phi_\varepsilon(x)=(e^{\varepsilon x}-1)/\varepsilon$, $\psi_\varepsilon(x)=e^{-\varepsilon x}\phi_\varepsilon(x)$, $q_\varepsilon=1/\psi_\varepsilon$, we define the rank-1 extremal block-diagonal split $L_{1,\varepsilon}(u,w) = \sqrt{L_{\mathrm{diag}}(u;\varepsilon)L_{\mathrm{diag}}(w;\varepsilon)}\succeq 0$ and $K_{1,\varepsilon}(s,t) = F_3(\min(s,t),\max(s,t);\varepsilon) - L_{1,\varepsilon}(\max(s,t),|t-s|)$. \[
K_{1,\varepsilon}(s,t) = \iint_{0<a<b<\infty} \Big(e^{s/6}\mathbf{1}_{[a,b]}(s)\Big)\Big(e^{t/6}\mathbf{1}_{[a,b]}(t)\Big)\,\rho(a,b;\varepsilon)\,\mathrm{d}a\,\mathrm{d}b.
\]

\section{General-$m$ Outlook and Remaining Frontier}

Douglas--Rachford SDP tests confirm that cyclic-shifted half-path PSD kernels (even $m=2r$, valid for indefinite $X$) and Schur-coupled path kernels (odd $m=2r+1$, valid for $X\ge 0$) exist to $10^{-11}$--$10^{-16}$ across $m=4$ ($n=0^+,1\dots 6,\infty$), $m=5$ ($n=0^+,1,2,\infty$), and $m=6$ ($n=0^+,1,2,\infty$). By finite-rank spectral approximation, all dimension-independent theorems above extend directly to trace-class positive operators on separable Hilbert spaces. The smallest remaining open cases of \href{https://openquantumproblems.com/p/M14}{OQP M14} are now $(5,4)/(4,5)$ and $(5,5)$.
